Swift Cheat Sheet

1. Variables & Constants
var name = "Omkar"      // Variable
let age = 21            // Constant

2. Data Types
var str: String = "Hello"
var num: Int = 10
var pi: Double = 3.14
var isOn: Bool = true

3. String Interpolation
let name = "Omkar"
print("Hello, \(name)")

4. Arrays
var fruits = ["Apple", "Banana"]
fruits.append("Mango")
print(fruits[0])

5. Dictionaries
var person = ["name": "Omkar", "age": "21"]
print(person["name"]!)

6. Sets
var numbers: Set = [1, 2, 3, 3]
print(numbers)   // {1, 2, 3}

7. If-Else
let score = 85
if score > 90 {
    print("A")
} else {
    print("B")
}

8. Switch
let day = 2
switch day {
case 1: print("Monday")
case 2: print("Tuesday")
default: print("Other day")
}

9. Loops
for i in 1...3 {
    print(i)
}

var n = 3
while n > 0 {
    print(n)
    n -= 1
}

10. Functions
func greet(name: String) -> String {
    return "Hello \(name)"
}
print(greet(name: "Omkar"))

11. Optionals
var name: String? = "Omkar"
print(name ?? "No Name") // Default value if nil

12. Optional Binding
if let realName = name {
    print(realName)
}

13. Classes & Objects
class Car {
    var brand = "Ford"
    func drive() { print("Driving") }
}
let myCar = Car()
myCar.drive()

14. Structs
struct Person {
    var name: String
}
let p = Person(name: "Omkar")
print(p.name)

15. Enums
enum Direction { case north, south }
let dir = Direction.north
print(dir)

16. Closures
let add = { (a: Int, b: Int) -> Int in
    return a + b
}
print(add(2, 3))

17. Guard
func check(age: Int?) {
    guard let a = age else { return }
    print("Age is \(a)")
}
check(age: 21)

18. Protocols
protocol Animal { func sound() }
class Dog: Animal {
    func sound() { print("Bark") }
}
Dog().sound()

19. Extensions
extension Int {
    func square() -> Int { return self * self }
}
print(4.square())

20. Error Handling
enum MyError: Error { case fail }
func risky() throws { throw MyError.fail }
do {
    try risky()
} catch {
    print("Error occurred")
}
